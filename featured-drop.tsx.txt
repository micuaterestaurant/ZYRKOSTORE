"use client"

import Image from "next/image"
import { useState } from "react"
import { products } from "@/lib/data"
import { MessageCircle, ChevronRight } from "lucide-react"

export function ProductCard({ product }: { product: typeof products[0] }) {
  const [selectedSize, setSelectedSize] = useState<string | null>(null)
  const [isHovered, setIsHovered] = useState(false)

  const whatsappUrl = `https://wa.me/34643574361?text=${encodeURIComponent(
    `Hola! Quiero pedir las ${product.name}${selectedSize ? ` en talla ${selectedSize}` : ""} (${product.price})`
  )}`

  return (
    <div
      className="product-card group bg-card rounded-2xl lg:rounded-3xl overflow-hidden border border-border/50 h-full"
      onMouseEnter={() => setIsHovered(true)}
      onMouseLeave={() => setIsHovered(false)}
    >
      {/* Image */}
      <div className="relative aspect-square overflow-hidden bg-secondary/30">
        {/* Tag */}
        <div className="absolute top-4 left-4 z-10">
          <span className="inline-block px-3 py-1.5 text-[10px] tracking-wider font-medium bg-foreground text-background rounded-full">
            {product.tag}
          </span>
        </div>

        {/* Product Image */}
        <div className="absolute inset-0 overflow-hidden">
          <Image
            src={product.image}
            alt={product.name}
            fill
            className={`object-cover w-full h-full transition-transform duration-500 ${
              isHovered ? "scale-105" : "scale-100"
            }`}
            sizes="(max-width: 768px) 50vw, (max-width: 1200px) 33vw, 25vw"
          />
        </div>

        {/* Overlay */}
        <div
          className={`absolute inset-0 bg-gradient-to-t from-black/50 via-transparent to-transparent transition-opacity duration-300 ${
            isHovered ? "opacity-100" : "opacity-0"
          }`}
        />

        {/* Desktop Size Selector */}
        <div
          className={`absolute bottom-4 left-4 right-4 hidden lg:block transition-all duration-300 ${
            isHovered ? "translate-y-0 opacity-100" : "translate-y-4 opacity-0"
          }`}
        >
          <p className="text-xs text-white/80 mb-2 tracking-wider">
            QUICK SELECT SIZE
          </p>

          <div className="flex flex-wrap gap-1.5">
            {product.sizes.slice(0, 6).map((size) => (
              <button
                key={size}
                onClick={() =>
                  setSelectedSize(size === selectedSize ? null : size)
                }
                className={`w-9 h-9 text-xs font-medium rounded-lg transition-all ${
                  selectedSize === size
                    ? "bg-white text-black"
                    : "bg-white/20 text-white hover:bg-white/30"
                }`}
              >
                {size}
              </button>
            ))}
          </div>
        </div>
      </div>

      {/* Content */}
      <div className="p-5 lg:p-6 flex flex-col h-full">
        {/* Name + Price */}
        <div className="flex items-start justify-between gap-3 mb-2">
          <h3 className="text-lg lg:text-xl font-semibold tracking-tight line-clamp-2 min-h-[56px] group-hover:text-foreground/80 transition-colors">
            {product.name}
          </h3>

          <span className="text-lg lg:text-xl font-bold whitespace-nowrap">
            {product.price}
          </span>
        </div>

        {/* Description */}
        <p className="text-sm text-muted-foreground mb-4 line-clamp-2 min-h-[40px]">
          {product.description}
        </p>

        {/* Colorway */}
        <p className="text-xs text-muted-foreground tracking-wider mb-4">
          {product.colorway}
        </p>

        {/* Mobile Size Selector */}
        <div className="lg:hidden mb-4">
          <p className="text-xs text-muted-foreground mb-2 tracking-wider">
            SELECT SIZE
          </p>

          <div className="flex flex-wrap gap-2">
            {product.sizes.map((size) => (
              <button
                key={size}
                onClick={() =>
                  setSelectedSize(size === selectedSize ? null : size)
                }
                className={`w-10 h-10 text-sm font-medium rounded-lg border transition-all ${
                  selectedSize === size
                    ? "bg-foreground text-background border-foreground"
                    : "border-border hover:border-foreground/50"
                }`}
              >
                {size}
              </button>
            ))}
          </div>
        </div>

        {/* CTA */}
        <a
          href={whatsappUrl}
          target="_blank"
          rel="noopener noreferrer"
          className="mt-auto flex items-center justify-center gap-2 w-full bg-foreground text-background py-3.5 rounded-xl font-medium text-sm tracking-wider hover:bg-foreground/90 transition-colors"
        >
          <MessageCircle className="w-4 h-4" />

          {selectedSize
            ? `ORDER SIZE ${selectedSize}`
            : "ORDER VIA WHATSAPP"}

          <ChevronRight className="w-4 h-4" />
        </a>
      </div>
    </div>
  )
}
